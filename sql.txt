-- SQL queries used in the project
-- ? represents a parameter passed from the backend

-- Table
CREATE TABLE IF NOT EXISTS Users(
   username VARCHAR(50) PRIMARY KEY,
   password_hash VARCHAR(255) NOT NULL,
   firstname VARCHAR(50) NOT NULL,
   lastname VARCHAR(50) NOT NULL,
   salary FLOAT,
   age INTEGER,
   registerday DATE,
   signintime DATETIME NULL
);

-- 1. User registration
INSERT INTO Users (username, password_hash, firstname, lastname, salary, age, registerday, signintime)
VALUES (?, ?, ?, ?, ?, ?, CURDATE(), NULL);

-- 2. User sign-in
-- password verification handled in backend
SELECT * FROM Users WHERE username = ?;
UPDATE Users SET signintime = NOW() WHERE username = ?;

-- 3. Search users by first and/or last name (a blank field is passed as NULL and ignored)
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users
WHERE (? IS NULL OR firstname = ?)
  AND (? IS NULL OR lastname = ?);

-- 4. Search users by userid
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users
WHERE username = ?;

-- 5. Users whose salary is between X and Y
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users
WHERE (? IS NULL OR salary >= ?)
  AND (? IS NULL OR salary <= ?);

-- 6. Users whose age is between X and Y
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users
WHERE (? IS NULL OR age >= ?)
  AND (? IS NULL OR age <= ?);

-- 7. Users who registered after a chosen username signed in
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users
WHERE registerday > (SELECT registerday FROM Users WHERE username = ?);

-- 8. Users who never signed in
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users
WHERE signintime IS NULL;

-- 9. Users who registered on the same day as a chosen username
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users
WHERE registerday = (SELECT registerday FROM Users WHERE username = ?)
  AND username <> ?;

-- 10. Users who registered today
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users
WHERE registerday = CURDATE();
